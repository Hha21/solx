The B200 budget for this round ran out before anything could be measured. My first step-0 probe got no response for 1800 s, and that wait used up the round's 20 minutes. After that, both the retried probe and `compile_b200` were refused with "the round's B200 budget (20 min) is used up". So the candidate below has not been compiled, run or timed. I wrote it from the design arithmetic and checked it by reading only.

### Rationale
This is lineage D's stored-h design, built as two plain launches on the same stream. The operator note bans PDL, so it is not used. K1 runs GEMM1 in FP16x3 on tcgen05 with half of W1 resident in shared memory. It applies GELU in the gelu2 form and writes h as fp16 hi plus a 2^11-scaled lo, 2 KB per token. It also writes per-tile sum-of-squares partials with plain stores. K2 sums those partials into the GRN scale for each batch and folds it into a resident W2'_b. It streams h through a 3-D TMA ring and runs GEMM2 in FP16x3 with a separate cross-term accumulator. Tiles line up with batch boundaries, so no tile spans two batches and there are no atomics, counters or zero-fills. K2 walks its tiles newest first, so it reads K1's last writes while they may still be in L2. Only rows with N ≥ 25088 tokens take this path; all other rows use r0.

```cuda file=kernel.cu
// 045 lineage D, r2 E4: two-launch stored-h FP16x3 tcgen05 pipeline for the L rows (N >= 25088), r0 SIMT elsewhere.
//   K1 (persistent, 2 CTAs per tile pair): CTA parity = hidden half; W1 half resident as fp16 hi/lo SW128 (128 KB).
//      Per batch-aligned 128-token tile: x hi/lo (converter warps, register prefetch), GEMM1 swap-AB FP16x3 into one
//      of two TMEM buffers, gelu2 epilogue writes h as fp16 hi + 2^11-scaled lo ([N][512] each, 2 KB/token) and
//      per-(tile, column half) channel sumsq partials with plain stores.
//   K2 (persistent, same pairing): CTA parity = output half. Builders reduce the partials of the current batch into
//      s = 1 + gw * G / (mean G + eps) and fold it into a resident W2'_b = W2 (.) s hi/lo image (128 KB, rebuilt when
//      the batch changes); TMA (3-D map, OOB rows zero-filled) streams h hi/lo k-atoms through a 3-stage ring;
//      GEMM2 main (hi*hi) and 2^11-scaled cross accumulators; out = main + cross/2048 + (b2 + W2.gb).
#include <cuda_runtime.h>
#include <cuda.h>

namespace r0 {
constexpr int D = 128, HID = 512, TM = 64, NC = 64, NT = 256;
constexpr int NCHUNK = HID / NC;
constexpr int LDK = D + 4;
constexpr int LDH = NC + 4;

__device__ __forceinline__ float gelu_erf(float v) { return v * 0.5f * (1.0f + erff(v * 0.7071067811865476f)); }

__device__ __forceinline__ void load_rows128(float* dst, const float* src, int nrows, int nvalid, int tid) {
  for (int i = tid; i < nrows * (D / 4); i += NT) {
    int r = i >> 5, k4 = (i & 31) << 2;
    float4 v = make_float4(0.f, 0.f, 0.f, 0.f);
    if (r < nvalid) v = __ldg(reinterpret_cast<const float4*>(src + (size_t)r * D + k4));
    *reinterpret_cast<float4*>(dst + r * LDK + k4) = v;
  }
}

__device__ __forceinline__ void gemm1(float (&acc)[4][4], const float* As, const float* Ws, int tm, int tn) {
#pragma unroll
  for (int i = 0; i < 4; ++i)
#pragma unroll
    for (int j = 0; j < 4; ++j) acc[i][j] = 0.f;
#pragma unroll 4
  for (int k = 0; k < D; k += 4) {
    float4 a[4], w[4];
#pragma unroll
    for (int i = 0; i < 4; ++i) a[i] = *reinterpret_cast<const float4*>(As + (tm * 4 + i) * LDK + k);
#pragma unroll
    for (int j = 0; j < 4; ++j) w[j] = *reinterpret_cast<const float4*>(Ws + (tn + 16 * j) * LDK + k);
#pragma unroll
    for (int i = 0; i < 4; ++i)
#pragma unroll
      for (int j = 0; j < 4; ++j) {
        acc[i][j] = fmaf(a[i].x, w[j].x, acc[i][j]);
        acc[i][j] = fmaf(a[i].y, w[j].y, acc[i][j]);
        acc[i][j] = fmaf(a[i].z, w[j].z, acc[i][j]);
        acc[i][j] = fmaf(a[i].w, w[j].w, acc[i][j]);
      }
  }
}

__global__ void __launch_bounds__(NT) k_stats(const float* __restrict__ x, const float* __restrict__ w1,
                                              const float* __restrict__ b1, float* __restrict__ part, int HW,
                                              int tiles) {
  extern __shared__ float4 smem4[];
  float* As = reinterpret_cast<float*>(smem4);
  float* Ws = As + TM * LDK;
  float* Red = Ws + NC * LDK;
  const int tile = blockIdx.x, b = blockIdx.y, tid = threadIdx.x;
  const int tm = tid >> 4, tn = tid & 15, warp = tid >> 5, lane = tid & 31;
  const int m0 = tile * TM, rows = min(TM, HW - m0);
  load_rows128(As, x + ((size_t)b * HW + m0) * D, TM, rows, tid);
  float* pout = part + ((size_t)b * tiles + tile) * HID;
  for (int c = 0; c < NCHUNK; ++c) {
    __syncthreads();
    load_rows128(Ws, w1 + (size_t)c * NC * D, NC, NC, tid);
    __syncthreads();
    float acc[4][4];
    gemm1(acc, As, Ws, tm, tn);
    float sq[4];
#pragma unroll
    for (int j = 0; j < 4; ++j) {
      float bias = __ldg(b1 + c * NC + tn + 16 * j);
      float s = 0.f;
#pragma unroll
      for (int i = 0; i < 4; ++i) {
        float h = gelu_erf(acc[i][j] + bias);
        if (tm * 4 + i < rows) s = fmaf(h, h, s);
      }
      s += __shfl_xor_sync(0xffffffffu, s, 16);
      sq[j] = s;
    }
    if (lane < 16) {
#pragma unroll
      for (int j = 0; j < 4; ++j) Red[warp * NC + tn + 16 * j] = sq[j];
    }
    __syncthreads();
    if (tid < NC) {
      float s = 0.f;
#pragma unroll
      for (int w = 0; w < NT / 32; ++w) s += Red[w * NC + tid];
      pout[c * NC + tid] = s;
    }
  }
}

__global__ void __launch_bounds__(HID) k_grn(const float* __restrict__ part, float* __restrict__ nf, int tiles,
                                             float eps) {
  __shared__ float red[HID / 32];
  const int b = blockIdx.x, c = threadIdx.x;
  const float* p = part + (size_t)b * tiles * HID + c;
  float s = 0.f;
  for (int t = 0; t < tiles; ++t) s += p[(size_t)t * HID];
  float g = __fsqrt_rn(s);
  float r = g;
#pragma unroll
  for (int o = 16; o > 0; o >>= 1) r += __shfl_xor_sync(0xffffffffu, r, o);
  if ((c & 31) == 0) red[c >> 5] = r;
  __syncthreads();
  float tot = 0.f;
#pragma unroll
  for (int w = 0; w < HID / 32; ++w) tot += red[w];
  float mean = __fdiv_rn(tot, (float)HID);
  nf[(size_t)b * HID + c] = __fdiv_rn(g, mean + eps);
}

__global__ void __launch_bounds__(NT) k_main(const float* __restrict__ x, const float* __restrict__ w1,
                                             const float* __restrict__ b1, const float* __restrict__ gw,
                                             const float* __restrict__ gb, const float* __restrict__ w2,
                                             const float* __restrict__ b2, const float* __restrict__ nf,
                                             float* __restrict__ out, int HW) {
  extern __shared__ float4 smem4[];
  float* As = reinterpret_cast<float*>(smem4);
  float* Ws = As + TM * LDK;
  float* Hs = Ws + D * LDH;
  const int tile = blockIdx.x, b = blockIdx.y, tid = threadIdx.x;
  const int tm = tid >> 4, tn = tid & 15;
  const int m0 = tile * TM, rows = min(TM, HW - m0);
  load_rows128(As, x + ((size_t)b * HW + m0) * D, TM, rows, tid);
  const float* nfb = nf + (size_t)b * HID;
  float acc2[4][8];
#pragma unroll
  for (int i = 0; i < 4; ++i)
#pragma unroll
    for (int j = 0; j < 8; ++j) acc2[i][j] = 0.f;
  for (int c = 0; c < NCHUNK; ++c) {
    __syncthreads();
    load_rows128(Ws, w1 + (size_t)c * NC * D, NC, NC, tid);
    __syncthreads();
    float acc[4][4];
    gemm1(acc, As, Ws, tm, tn);
#pragma unroll
    for (int j = 0; j < 4; ++j) {
      const int n = c * NC + tn + 16 * j;
      const float bias = __ldg(b1 + n), g = __ldg(gw + n), be = __ldg(gb + n), f = __ldg(nfb + n);
#pragma unroll
      for (int i = 0; i < 4; ++i) {
        float h = gelu_erf(acc[i][j] + bias);
        Hs[(tm * 4 + i) * LDH + tn + 16 * j] = g * (h * f) + be + h;
      }
    }
    __syncthreads();
    for (int i = tid; i < D * (NC / 4); i += NT) {
      int o = i >> 4, k4 = (i & 15) << 2;
      *reinterpret_cast<float4*>(Ws + o * LDH + k4) =
          __ldg(reinterpret_cast<const float4*>(w2 + (size_t)o * HID + c * NC + k4));
    }
    __syncthreads();
#pragma unroll 2
    for (int k = 0; k < NC; k += 4) {
      float4 a[4];
#pragma unroll
      for (int i = 0; i < 4; ++i) a[i] = *reinterpret_cast<const float4*>(Hs + (tm * 4 + i) * LDH + k);
#pragma unroll
      for (int j = 0; j < 8; ++j) {
        float4 w = *reinterpret_cast<const float4*>(Ws + (tn + 16 * j) * LDH + k);
#pragma unroll
        for (int i = 0; i < 4; ++i) {
          acc2[i][j] = fmaf(a[i].x, w.x, acc2[i][j]);
          acc2[i][j] = fmaf(a[i].y, w.y, acc2[i][j]);
          acc2[i][j] = fmaf(a[i].z, w.z, acc2[i][j]);
          acc2[i][j] = fmaf(a[i].w, w.w, acc2[i][j]);
        }
      }
    }
  }
  float* ob = out + ((size_t)b * HW + m0) * D;
#pragma unroll
  for (int j = 0; j < 8; ++j) {
    const int o = tn + 16 * j;
    const float bias = __ldg(b2 + o);
#pragma unroll
    for (int i = 0; i < 4; ++i)
      if (tm * 4 + i < rows) ob[(size_t)(tm * 4 + i) * D + o] = acc2[i][j] + bias;
  }
}

constexpr int SMEM_STATS = (TM * LDK + NC * LDK + (NT / 32) * NC) * 4;
constexpr int SMEM_MAIN = (TM * LDK + D * LDH + TM * LDH) * 4;
}  // namespace r0

extern "C" void launch_grn_mlp(const float* x, const float* w1, const float* b1, const float* gw, const float* gb,

  k_main<<<grid, NT, SMEM_MAIN, stream>>>(x, w1, b1, gw, gb, w2, b2, nf, out, HW);
}

// GELU (erf form) on a pair via Abramowitz-Stegun 7.1.26 erfc (|err| <= 1.5e-7); ss += h^2 per lane of the pair
__device__ __forceinline__ void gelu2(float& x0, float& x1, uint64_t& ss2) {
  float t0, t1, e0, e1, q0, q1, d0, d1;
  asm("rcp.approx.ftz.f32 %0, %1;" : "=f"(t0) : "f"(fmaf(fabsf(x0), 0.231641900f, 1.f)));
  asm("rcp.approx.ftz.f32 %0, %1;" : "=f"(t1) : "f"(fmaf(fabsf(x1), 0.231641900f, 1.f)));
  const uint64_t T = f2pack(t0, t1), X = f2pack(x0, x1);
  uint64_t P = fma2(T, f2pack(0.5307027145f, 0.5307027145f), f2pack(-0.7265760135f, -0.7265760135f));
  P = fma2(T, P, f2pack(0.7107068705f, 0.7107068705f));
  P = fma2(T, P, f2pack(-0.142248368f, -0.142248368f));
  P = fma2(T, P, f2pack(0.127414796f, 0.127414796f));
  P = mul2(P, T);
  f2unpack(mul2(X, mul2(X, f2pack(-0.72134752044f, -0.72134752044f))), q0, q1);
  asm("ex2.approx.ftz.f32 %0, %1;" : "=f"(e0) : "f"(q0));
  asm("ex2.approx.ftz.f32 %0, %1;" : "=f"(e1) : "f"(q1));
  const uint64_t XH = mul2(X, mul2(P, f2pack(e0, e1)));
  f2unpack(fma2(XH, f2pack(-2.f, -2.f), X), d0, d1);
  const uint64_t H = add2(XH, f2pack(fmaxf(d0, 0.f), fmaxf(d1, 0.f)));
  ss2 = fma2(H, H, ss2);
  f2unpack(H, x0, x1);
}

// ---------------------------------------------- K1: GEMM1 + GELU, stores h hi/lo and sumsq partials
// converter thread ct (0..255): chunk task i -> token n = (ct + 256 i) >> 4, 8-float chunk c = (ct + 256 i) & 15
__device__ __forceinline__ void k1_load_x(float4 (&xr)[16], const K1P& p, int g, int ct) {
  const int b = g / p.TB, tb = g - b * p.TB;
  const int nv = min(128, p.HW - tb * 128);
  const float* base = p.x + ((size_t)b * p.HW + (size_t)tb * 128) * 128;
#pragma unroll
  for (int i = 0; i < 8; i++) {
    const int id = ct + 256 * i, n = id >> 4, c = id & 15;
    if (n < nv) {
      const float4* s = reinterpret_cast<const float4*>(base + (size_t)n * 128 + c * 8);
      xr[2 * i] = __ldg(s);
      xr[2 * i + 1] = __ldg(s + 1);
    } else {
      xr[2 * i] = make_float4(0.f, 0.f, 0.f, 0.f);
      xr[2 * i + 1] = xr[2 * i];
    }
  }
}

__global__ void __launch_bounds__(K1_NT, 1) k1_gemm1(const K1P p) {
  extern __shared__ __align__(1024) unsigned char smraw[];
  unsigned char* sm = (unsigned char*)(((uintptr_t)smraw + 1023) & ~(uintptr_t)1023);
  const uint32_t sb = s_addr(sm);
  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int hh = blockIdx.x & 1, pair = blockIdx.x >> 1, np = gridDim.x >> 1;
  const int g0 = (int)((long long)pair * p.ntiles / np), g1 = (int)((long long)(pair + 1) * p.ntiles / np);
  const int nt = g1 - g0;
  const uint32_t bar_xfull = sb + K1_BAR, bar_xfree = bar_xfull + 8, bar_tfull = bar_xfull + 16, bar_tfree = bar_xfull + 32;
  uint32_t* s_tmem = (uint32_t*)(sm + K1_BAR + 64);
  if (tid == 0) {
    mbar_init(bar_xfull, 256); mbar_init(bar_xfree, 1);
    mbar_init(bar_tfull, 1); mbar_init(bar_tfull + 8, 1);
    mbar_init(bar_tfree, 256); mbar_init(bar_tfree + 8, 256);
    asm volatile("fence.mbarrier_init.release.cluster;" ::: "memory");
  }
  if (warp == 16) {
    asm volatile("tcgen05.alloc.cta_group::1.sync.aligned.shared::cta.b32 [%0], %1;" ::"r"(s_addr(s_tmem)), "r"(512) : "memory");
    asm volatile("tcgen05.relinquish_alloc_permit.cta_group::1.sync.aligned;" ::: "memory");
  }
  float4 xr[16];
  const bool is_conv = (warp >= 8 && warp < 16);
  const int ct = tid - 256;
  if (is_conv) {
    k1_load_x(xr, p, g0, ct);
  } else {
    // W1 half image: row r (0..255) of M block r>>7, TMEM lane r&127 <-> hidden channel hh*256 + 2*(r&127) + (r>>7)
    const int bi = (tid < 256) ? tid : tid - 256;
#pragma unroll 2
    for (int t = bi; t < 4096; t += 288) {
      const int r = t >> 4, c = t & 15;
      const int ch = hh * 256 + 2 * (r & 127) + (r >> 7);
      const float4* s = reinterpret_cast<const float4*>(p.w1 + (size_t)ch * 128 + c * 8);
      uint4 hi, lo;
      split8(__ldg(s), __ldg(s + 1), 1.f, hi, lo);
      const uint32_t off = (uint32_t)(c >> 3) * 32768u + (uint32_t)r * 128u + ((uint32_t)((c & 7) ^ (r & 7)) << 4);
      *reinterpret_cast<uint4*>(sm + K1_W1H + off) = hi;
      *reinterpret_cast<uint4*>(sm + K1_W1L + off) = lo;
    }
    fence_proxy_async();
  }
  tc_before();
  __syncthreads();
  tc_after();
  const uint32_t tmem = *s_tmem;

  if (is_conv) {
    // ---- x converters: fp32 x tile (registers) -> hi/lo K-major SW128 B image; prefetch the next tile
#pragma unroll 1
    for (int j = 0; j < nt; j++) {
      if (j > 0) mbar_wait(bar_xfree, (j - 1) & 1);
#pragma unroll
      for (int i = 0; i < 8; i++) {
        const int id = ct + 256 * i, n = id >> 4, c = id & 15;
        uint4 hi, lo;
        split8(xr[2 * i], xr[2 * i + 1], 1.f, hi, lo);
        const uint32_t off = (uint32_t)(c >> 3) * 16384u + (uint32_t)n * 128u + ((uint32_t)((c & 7) ^ (n & 7)) << 4);
        *reinterpret_cast<uint4*>(sm + K1_XH + off) = hi;
        *reinterpret_cast<uint4*>(sm + K1_XL + off) = lo;
      }
      fence_proxy_async();
      mbar_arrive(bar_xfull);
      if (j + 1 < nt) k1_load_x(xr, p, g0 + j + 1, ct);
    }
  } else if (warp == 16) {
    // ---- MMA warp: D[buf][m] (hidden 128 x token 128) = W1 block m * x^T, FP16x3, single fp32 accumulator
    const uint32_t id = idesc_f16(128);
#pragma unroll 1
    for (int j = 0; j < nt; j++) {
      const int buf = j & 1;
      mbar_wait(bar_xfull, j & 1);
      if (j >= 2) mbar_wait(bar_tfree + 8 * buf, ((j >> 1) - 1) & 1);
      tc_after();
#pragma unroll
      for (int m = 0; m < 2; m++) {
        const uint32_t d = tmem + buf * 256 + m * 128;
#pragma unroll
        for (int st = 0; st < 8; st++) {
          const uint32_t a = st >> 2, kk = st & 3;
          const uint64_t aH = sdesc(sb + K1_W1H + a * 32768 + m * 16384 + kk * 32);
          const uint64_t aL = sdesc(sb + K1_W1L + a * 32768 + m * 16384 + kk * 32);
          const uint64_t bH = sdesc(sb + K1_XH + a * 16384 + kk * 32);
          const uint64_t bL = sdesc(sb + K1_XL + a * 16384 + kk * 32);
          mma_issue(d, aH, bL, id, st > 0 ? 1u : 0u);
          mma_issue(d, aL, bH, id, 1u);
          mma_issue(d, aH, bH, id, 1u);
        }
      }
      mma_commit(bar_xfree);
      mma_commit(bar_tfull + 8 * buf);
    }
  } else if (warp < 8) {
    // ---- epilogue: thread = channel pair (c0, c0 + 1), warp = (lane quadrant, 64-token column half)
    const int q = warp & 3, chh = warp >> 2, L = 32 * q + lane;
    const int c0 = hh * 256 + 2 * L;
    const float bA = __ldg(p.b1 + c0), bB = __ldg(p.b1 + c0 + 1);
    const uint64_t bAB = f2pack(bA, bB);
#pragma unroll 1
    for (int j = 0; j < nt; j++) {
      const int g = g0 + j, buf = j & 1;
      const int b = g / p.TB, tb = g - b * p.TB;
      const int nv = min(128, p.HW - tb * 128);
      const size_t tok0 = (size_t)b * p.HW + (size_t)tb * 128;
      mbar_wait(bar_tfull + 8 * buf, (j >> 1) & 1);
      tc_after();
      uint64_t ss2 = f2pack(0.f, 0.f);
#pragma unroll 1
      for (int cc = 0; cc < 4; cc++) {
        const uint32_t ta = tmem + ((uint32_t)(32 * q) << 16) + buf * 256 + chh * 64 + cc * 16;
        uint32_t r0[16], r1[16];
        tmem_ld16(ta, r0);
        tmem_ld16(ta + 128, r1);
        tmem_wait_ld();
        pin16(r0);
        pin16(r1);
        if (cc == 3) {  // TMEM buffer free: only the TMEM reads are handed over, so a relaxed arrive is enough
          tc_before();
          mbar_arrive_relaxed(bar_tfree + 8 * buf);
        }
#pragma unroll
        for (int e = 0; e < 16; e++) {
          const int t = chh * 64 + cc * 16 + e;
          const bool ok = t < nv;
          float v0, v1;
          f2unpack(add2(f2pack(__uint_as_float(r0[e]), __uint_as_float(r1[e])), bAB), v0, v1);
          v0 = ok ? v0 : 0.f;
          v1 = ok ? v1 : 0.f;
          gelu2(v0, v1, ss2);
          uint32_t hi, lo;
          split2(v0, v1, 2048.f, hi, lo);
          if (ok) {
            const size_t idx = (tok0 + t) * 256 + (size_t)(hh * 128 + L);
            p.hhi[idx] = hi;
            p.hlo[idx] = lo;
          }
        }
      }
      float s0, s1;
      f2unpack(ss2, s0, s1);
      *reinterpret_cast<float2*>(p.part + ((size_t)g * 2 + chh) * 512 + c0) = make_float2(s0, s1);
    }
  }
  tc_before();
  __syncthreads();
  if (warp == 16) {
    tc_after();
    asm volatile("tcgen05.dealloc.cta_group::1.sync.aligned.b32 %0, %1;" ::"r"(tmem), "r"(512) : "memory");
  }
}

// ---------------------------------------------- K2: GRN fold + GEMM2 over stored h
__global__ void __launch_bounds__(K2_NT, 1) k2_gemm2(const __grid_constant__ CUtensorMap tmh, const __grid_constant__ CUtensorMap tml,
                                                    const K2P p) {
  extern __shared__ __align__(1024) unsigned char smraw[];
  unsigned char* sm = (unsigned char*)(((uintptr_t)smraw + 1023) & ~(uintptr_t)1023);
  if ((uintptr_t)(sm - smraw) + K2_END > K2_SMEM) __trap();
  const uint32_t sb = s_addr(sm);
  const int tid = threadIdx.x, warp = tid >> 5, lane = tid & 31;
  const int oh = blockIdx.x & 1, pair = blockIdx.x >> 1, np = gridDim.x >> 1;
  const int g0 = (int)((long long)pair * p.ntiles / np), g1 = (int)((long long)(pair + 1) * p.ntiles / np);
  const int nt = g1 - g0;  // tile j of this CTA is g1 - 1 - j (newest h first: the most likely to still sit in L2)
  const uint32_t bar_full = sb + K2_BAR, bar_empty = bar_full + 24, bar_tfull = bar_full + 48, bar_tfree = bar_full + 64,
                 bar_wfull = bar_full + 80, bar_wfree = bar_full + 88;
  uint32_t* s_tmem = (uint32_t*)(sm + K2_BAR + 112);
  float* s_s = (float*)(sm + K2_S);
  float* s_bias = (float*)(sm + K2_BIAS);
  float* s_red = (float*)(sm + K2_RED);
  if (tid == 0) {
    for (int i = 0; i < NST; i++) { mbar_init(bar_full + 8 * i, 1); mbar_init(bar_empty + 8 * i, 1); }
    mbar_init(bar_tfull, 1); mbar_init(bar_tfull + 8, 1);
    mbar_init(bar_tfree, 256); mbar_init(bar_tfree + 8, 256);
    mbar_init(bar_wfull, 256); mbar_init(bar_wfree, 1);
    asm volatile("fence.mbarrier_init.release.cluster;" ::: "memory");
  }
  if (warp == 17) {
    asm volatile("tcgen05.alloc.cta_group::1.sync.aligned.shared::cta.b32 [%0], %1;" ::"r"(s_addr(s_tmem)), "r"(256) : "memory");
    asm volatile("tcgen05.relinquish_alloc_permit.cta_group::1.sync.aligned;" ::: "memory");
  }
  // bias' = b2 + W2 . gb for this CTA's 64 outputs
  for (int o = warp; o < 64; o += K2_NT / 32) {
    const float4* wr = reinterpret_cast<const float4*>(p.w2 + (size_t)(oh * 64 + o) * 512);
    const float4* gr = reinterpret_cast<const float4*>(p.gb);
    float d = 0.f;
#pragma unroll
    for (int i = 0; i < 4; i++) {
      const float4 w = __ldg(wr + lane + 32 * i), g = __ldg(gr + lane + 32 * i);
      d = fmaf(w.x, g.x, fmaf(w.y, g.y, fmaf(w.z, g.z, fmaf(w.w, g.w, d))));
    }
    d = warp_sum(d);
    if (lane == 0) s_bias[o] = __ldg(p.b2 + oh * 64 + o) + d;
  }
  tc_before();
  __syncthreads();
  tc_after();
  const uint32_t tmem = *s_tmem;

  if (warp == 16) {
    // ---- TMA producer: per tile, 8 k-atoms of (h hi, h lo) [128 tokens][64 k] SW128, 32 KB per stage
    int cnt = 0;
#pragma unroll 1
    for (int j = 0; j < nt; j++) {
      const int g = g1 - 1 - j, b = g / p.TB, tb = g - b * p.TB;
#pragma unroll 1
      for (int a = 0; a < 8; a++, cnt++) {
        const int s = cnt % NST;
        if (cnt >= NST) mbar_wait(bar_empty + 8 * s, ((cnt / NST) - 1) & 1);
        if (lane == 0) {
          const uint32_t dst = sb + K2_RING + s * 32768;
          mbar_expect(bar_full + 8 * s, 32768);
          tma3d(dst, &tmh, a * 64, tb * 128, b, bar_full + 8 * s);
          tma3d(dst + 16384, &tml, a * 64, tb * 128, b, bar_full + 8 * s);
        }
        __syncwarp();
      }
    }
  } else if (warp == 17) {
    // ---- MMA warp: Dm[token][out] = hhi W2'hi^T; Dc = (hhi W2'lo^T + hlo W2'hi^T), lo parts scaled by 2^11
    const uint32_t id = idesc_f16(64);
    int cnt = 0, seg = -1, prev_b = -1;
#pragma unroll 1
    for (int j = 0; j < nt; j++) {
      const int g = g1 - 1 - j, b = g / p.TB, buf = j & 1;
      if (j >= 2) mbar_wait(bar_tfree + 8 * buf, ((j >> 1) - 1) & 1);
      if (b != prev_b) { seg++; mbar_wait(bar_wfull, seg & 1); prev_b = b; }
      tc_after();
      const uint32_t dm = tmem + buf * 128, dc = dm + 64;
#pragma unroll 1
      for (int a = 0; a < 8; a++, cnt++) {
        const int s = cnt % NST;
        mbar_wait(bar_full + 8 * s, (cnt / NST) & 1);
        tc_after();
#pragma unroll
        for (int kk = 0; kk < 4; kk++) {
          const uint64_t aH = sdesc(sb + K2_RING + s * 32768 + kk * 32);
          const uint64_t aL = sdesc(sb + K2_RING + s * 32768 + 16384 + kk * 32);
          const uint64_t bH = sdesc(sb + K2_WH + a * 8192 + kk * 32);
          const uint64_t bL = sdesc(sb + K2_WL + a * 8192 + kk * 32);
          const uint32_t acc = (a | kk) ? 1u : 0u;
          mma_issue(dm, aH, bH, id, acc);
          mma_issue(dc, aH, bL, id, acc);
          mma_issue(dc, aL, bH, id, 1u);
        }
        mma_commit(bar_empty + 8 * s);
      }
      mma_commit(bar_tfull + 8 * buf);
      const bool seg_end = (j + 1 == nt) || ((g - 1) / p.TB != b);
      if (seg_end && j + 1 < nt) mma_commit(bar_wfree);
    }
  } else if (warp >= 8) {
    // ---- W2' builders: per batch segment, s = 1 + gw * G / (mean G + eps), W2'_b = W2 (.) s as hi / 2^11 lo
    const int bt = tid - 256;
    const int cch = bt & 63, r0 = bt >> 6;
    int seg = 0, j = 0;
#pragma unroll 1
    while (j < nt) {
      const int b = (g1 - 1 - j) / p.TB;
      int je = j;
      while (je < nt && (g1 - 1 - je) / p.TB == b) je++;
      float S0 = 0.f, S1 = 0.f;
      const float* pp = p.part + (size_t)b * p.TB * 1024 + 2 * bt;
#pragma unroll 4
      for (int t = 0; t < 2 * p.TB; t++) {
        const float2 v = __ldcg(reinterpret_cast<const float2*>(pp + (size_t)t * 512));
        S0 += v.x;
        S1 += v.y;
      }
      const float G0 = __fsqrt_rn(S0), G1 = __fsqrt_rn(S1);
      const float ws = warp_sum(G0 + G1);
      if (lane == 0) s_red[warp - 8] = ws;
      asm volatile("bar.sync 1, 256;" ::: "memory");
      float tot = 0.f;
#pragma unroll
      for (int i = 0; i < 8; i++) tot += s_red[i];
      const float den = __fdiv_rn(tot, 512.f) + p.eps;
      s_s[2 * bt] = fmaf(__ldg(p.gw + 2 * bt), __fdiv_rn(G0, den), 1.f);
      s_s[2 * bt + 1] = fmaf(__ldg(p.gw + 2 * bt + 1), __fdiv_rn(G1, den), 1.f);
      asm volatile("bar.sync 1, 256;" ::: "memory");
      float sv[8];
#pragma unroll
      for (int i = 0; i < 8; i++) sv[i] = s_s[8 * cch + i];
      if (seg > 0) mbar_wait(bar_wfree, (seg - 1) & 1);
#pragma unroll 1
      for (int ii = 0; ii < 4; ii++) {
        float4 wv[8];
#pragma unroll
        for (int i = 0; i < 4; i++) {
          const int r = r0 + 4 * (4 * ii + i);
          const float4* src = reinterpret_cast<const float4*>(p.w2 + (size_t)(oh * 64 + r) * 512 + 8 * cch);
          wv[2 * i] = __ldg(src);
          wv[2 * i + 1] = __ldg(src + 1);
        }
#pragma unroll
        for (int i = 0; i < 4; i++) {
          const int r = r0 + 4 * (4 * ii + i);
          float4 u = wv[2 * i], v = wv[2 * i + 1];
          u.x *= sv[0]; u.y *= sv[1]; u.z *= sv[2]; u.w *= sv[3];
          v.x *= sv[4]; v.y *= sv[5]; v.z *= sv[6]; v.w *= sv[7];
          uint4 hi, lo;
          split8(u, v, 2048.f, hi, lo);
          const uint32_t off = (uint32_t)(cch >> 3) * 8192u + (uint32_t)r * 128u + ((uint32_t)((cch & 7) ^ (r & 7)) << 4);
          *reinterpret_cast<uint4*>(sm + K2_WH + off) = hi;
          *reinterpret_cast<uint4*>(sm + K2_WL + off) = lo;
        }
      }
      fence_proxy_async();
      mbar_arrive(bar_wfull);
      seg++;
      j = je;
    }
  } else {
    // ---- epilogue: thread = token row (lane quadrant), warp half oc = 32 outputs
    const int q = warp & 3, oc = warp >> 2;
#pragma unroll 1
    for (int j = 0; j < nt; j++) {
      const int g = g1 - 1 - j, b = g / p.TB, tb = g - b * p.TB, buf = j & 1;
      mbar_wait(bar_tfull + 8 * buf, (j >> 1) & 1);
      tc_after();
      const uint32_t ta = tmem + ((uint32_t)(32 * q) << 16) + buf * 128 + 32 * oc;
      uint32_t m0[16], m1[16], x0[16], x1[16];
      tmem_ld16(ta, m0);
      tmem_ld16(ta + 16, m1);
      tmem_ld16(ta + 64, x0);
      tmem_ld16(ta + 80, x1);
      tmem_wait_ld();
      pin16(m0); pin16(m1); pin16(x0); pin16(x1);
      tc_before();
      mbar_arrive_relaxed(bar_tfree + 8 * buf);
      const int row = 32 * q + lane;
      if (tb * 128 + row < p.HW) {
        float* dst = p.out + ((size_t)b * p.HW + (size_t)tb * 128 + row) * 128 + oh * 64 + 32 * oc;
        const float* bi = s_bias + 32 * oc;
        constexpr float inv = 1.f / 2048.f;
#pragma unroll
        for (int i = 0; i < 4; i++) {
          float4 o0, o1;
          o0.x = fmaf(__uint_as_float(x0[4 * i + 0]), inv, __uint_as_float(m0[4 * i + 0]) + bi[4 * i + 0]);
          o0.y = fmaf(__uint_as_float(x0[4 * i + 1]), inv, __uint_as_float(m0[4 * i + 1]) + bi[4 * i + 1]);
          o0.z = fmaf(__uint_as_float(x0[4 * i + 2]), inv, __uint_as_float(m0[4 * i + 2]) + bi[4 * i + 2]);
          o0.w = fmaf(__uint_as_float(x0[4 * i + 3]), inv, __uint_as_float(m0[4 * i + 3]) + bi[4 * i + 3]);
          o1.x = fmaf(__uint_as_float(x1[4 * i + 0]), inv, __uint_as_float(m1[4 * i + 0]) + bi[16 + 4 * i + 0]);
          o1.y = fmaf(__uint_as_float(x1[4 * i + 1]), inv, __uint_as_float(m1[4 * i + 1]) + bi[16 + 4 * i + 1]);
          o1.z = fmaf(__uint_as_float(x1[4 * i + 2]), inv, __uint_as_float(m1[4 * i + 2]) + bi[16 + 4 * i + 2]);
          o1.w = fmaf(__uint_as_float(x1[4 * i + 3]), inv, __uint_as_float(m1[4 * i + 3]) + bi[16 + 4 * i + 3]);
          *reinterpret_cast<float4*>(dst + 4 * i) = o0;
          *reinterpret_cast<float4*>(dst + 16 + 4 * i) = o1;
        }
      }
    }
  }
  tc_before();
  __syncthreads();
  if (warp == 17) {
    tc_after();
    asm volatile("tcgen05.dealloc.cta_group::1.sync.aligned.b32 %0, %1;" ::"r"(tmem), "r"(256) : "memory");
  }
}

extern "C" size_t sh_workspace_bytes(int B, int HW) {
  const size_t N = (size_t)B * HW;
  const size_t TB = (HW + 127) / 128;
  return 2 * N * 1024 + (size_t)B * TB * 2 * 512 * 4;
}

static const char* encode_h_map(CUtensorMap* m, void* ptr, int B, int HW) {
  cuuint64_t dims[3] = {512, (cuuint64_t)HW, (cuuint64_t)B};
  cuuint64_t strides[2] = {1024, (cuuint64_t)HW * 1024};
  cuuint32_t box[3] = {64, 128, 1};
  cuuint32_t es[3] = {1, 1, 1};
  CUresult r = cuTensorMapEncodeTiled(m, CU_TENSOR_MAP_DATA_TYPE_FLOAT16, 3, ptr, dims, strides, box, es,
                                      CU_TENSOR_MAP_INTERLEAVE_NONE, CU_TENSOR_MAP_SWIZZLE_128B,
                                      CU_TENSOR_MAP_L2_PROMOTION_L2_256B, CU_TENSOR_MAP_FLOAT_OOB_FILL_NONE);
  return r == CUDA_SUCCESS ? nullptr : "cuTensorMapEncodeTiled failed";
}

extern "C" const char* launch_sh(const float* x, const float* w1, const float* b1, const float* gw, const float* gb,
                                 const float* w2, const float* b2, float eps, float* out, void* ws, int B, int HW,
                                 cudaStream_t stream) {
  using namespace sh;
  static bool attrs_set = false;
  cudaError_t e;
  if (!attrs_set) {  // one maximal value for every shape
    e = cudaFuncSetAttribute(k1_gemm1, cudaFuncAttributeMaxDynamicSharedMemorySize, (int)K1_SMEM);
    if (e != cudaSuccess) return cudaGetErrorString(e);
    e = cudaFuncSetAttribute(k2_gemm2, cudaFuncAttributeMaxDynamicSharedMemorySize, (int)K2_SMEM);
    if (e != cudaSuccess) return cudaGetErrorString(e);
    attrs_set = true;
  }
  const size_t N = (size_t)B * HW;
  const int TB = (HW + 127) / 128, ntiles = B * TB;
  const int np = ntiles < 74 ? ntiles : 74;
  unsigned char* wsb = (unsigned char*)ws;
  uint32_t* hhi = (uint32_t*)wsb;
  uint32_t* hlo = (uint32_t*)(wsb + N * 1024);
  float* part = (float*)(wsb + 2 * N * 1024);
  CUtensorMap mh, ml;
  const char* err = encode_h_map(&mh, hhi, B, HW);
  if (err) return err;
  err = encode_h_map(&ml, hlo, B, HW);
  if (err) return err;
  K1P p1{x, w1, b1, hhi, hlo, part, B, HW, TB, ntiles};
  K2P p2{w2, b2, gw, gb, part, out, B, HW, TB, ntiles, eps};
  k1_gemm1<<<2 * np, K1_NT, K1_SMEM, stream>>>(p1);
  e = cudaGetLastError();
  if (e != cudaSuccess) return cudaGetErrorString(e);
  k2_gemm2<<<2 * np, K2_NT, K2_SMEM, stream>>>(mh, ml, p2);
  e = cudaGetLastError();
  if (e != cudaSuccess) return cudaGetErrorString(e);
  return nullptr;
```

```cpp file=binding.cpp
#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>

extern "C" void launch_grn_mlp(const float* x, const float* w1, const float* b1, const float* gw, const float* gb,
                               const float* w2, const float* b2, float eps, float* out, float* part, float* nf,
                               int B, int HW, cudaStream_t stream);
extern "C" size_t sh_workspace_bytes(int B, int HW);
extern "C" const char* launch_sh(const float* x, const float* w1, const float* b1, const float* gw, const float* gb,
                                 const float* w2, const float* b2, float eps, float* out, void* ws, int B, int HW,
                                 cudaStream_t stream);

// stored-h two-launch path for the L rows (B*HW >= 25088 tokens); r0 SIMT for the others
static constexpr int64_t kShMinTokens = 25088;

void run(torch::Tensor hidden_states, torch::Tensor pwconv1_weight, torch::Tensor pwconv1_bias,
         torch::Tensor grn_weight, torch::Tensor grn_bias, torch::Tensor pwconv2_weight, torch::Tensor pwconv2_bias,
         double eps, torch::Tensor output) {
  TORCH_CHECK(hidden_states.is_cuda() && hidden_states.scalar_type() == torch::kFloat32, "fp32 CUDA input");
  TORCH_CHECK(hidden_states.dim() == 4 && hidden_states.size(3) == 128, "[B,H,W,128]");
  TORCH_CHECK(pwconv1_weight.size(0) == 512 && pwconv1_weight.size(1) == 128, "W1 [512,128]");
  TORCH_CHECK(pwconv2_weight.size(0) == 128 && pwconv2_weight.size(1) == 512, "W2 [128,512]");
  for (auto* t : {&hidden_states, &pwconv1_weight, &pwconv1_bias, &grn_weight, &grn_bias, &pwconv2_weight,
                  &pwconv2_bias, &output})
    TORCH_CHECK(t->is_contiguous(), "contiguous tensors required");
  const c10::cuda::CUDAGuard guard(hidden_states.device());
  const int B = (int)hidden_states.size(0);
  const int HW = (int)(hidden_states.size(1) * hidden_states.size(2));
  cudaStream_t stream = at::cuda::getCurrentCUDAStream();
  if ((int64_t)B * HW >= kShMinTokens && HW >= 128) {
    auto ws = torch::empty({(int64_t)sh_workspace_bytes(B, HW)}, hidden_states.options().dtype(torch::kUInt8));
    const char* err = launch_sh(hidden_states.data_ptr<float>(), pwconv1_weight.data_ptr<float>(),
                                pwconv1_bias.data_ptr<float>(), grn_weight.data_ptr<float>(), grn_bias.data_ptr<float>(),
                                pwconv2_weight.data_ptr<float>(), pwconv2_bias.data_ptr<float>(), (float)eps,
                                output.data_ptr<float>(), ws.data_ptr(), B, HW, stream);
    TORCH_CHECK(err == nullptr, err ? err : "");
    return;
  }
  const int tiles = (HW + 63) / 64;
  auto part = torch::empty({(int64_t)B * tiles * 512}, hidden_states.options());
  auto nf = torch::empty({(int64_t)B * 512}, hidden_states.options());
  launch_grn_mlp(hidden_states.data_ptr<float>(), pwconv1_weight.data_ptr<float>(), pwconv1_bias.data_ptr<float>(),
                 grn_weight.data_ptr<float>(), grn_bias.data_ptr<float>(), pwconv2_weight.data_ptr<float>(),
                 pwconv2_bias.data_ptr<float>(), (float)eps, output.data_ptr<float>(), part.data_ptr<float>(),
                 nf.data_ptr<float>(), B, HW, stream);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) { m.def("run", &run, "fused linear-gelu-grn-linear"); }
```
