### Rationale
This is one persistent cooperative launch for experiment E3. It runs 148 CTAs at one per SM and gives each CTA one 128-token tile per round, in batch-major order. Both GEMMs are FP16x3 on tcgen05, and GEMM1 is laid out hidden-by-token so the GRN sum of squares is a per-thread sum.

Each round does two things:
- **GEMM1 on this round's tile:** bias and GELU, then the sum of squares is added per batch with `red.add`. The fp32 h goes into this CTA's two-slot workspace ring with an L2 evict_last hint.
- **GEMM2 on last round's tile:** wait for that tile's batch counters, read h back from L2, scale and split it into fp16 hi/lo in shared memory. GEMM2 keeps a separate accumulator for the cross terms.

The hi/lo weight images are built in phase 0 and streamed through a 4-stage `cp.async.bulk` ring. Every size runs the same code; at S/M it is one round plus a drain round.

// schedule t = r * grid + cta). Round r: GEMM1 (swap-AB, D1[hidden][token], FP16x3) of tile r -> + b1, GELU, per-batch
// sumsq red.add, fp32 h into this CTA's workspace slot (r & 1) with L2 evict_last, per-batch token counters. Then GEMM2 of
// tile r - 1: wait for its batches' counters, GRN scale s_b = 1 + gw * nf, h' = h * s re-landed from L2 and split hi/lo
// into the MN-major smem B operand, GEMM2 (D2[out][token]) with a separate hi.lo + lo.hi accumulator, out = D2 + b2'.
// b2' = b2 + W2.gb. Weights: phase 0 writes fp16 hi/lo SW128 byte images (512 KB) into the workspace; a producer warp

// exact-class GELU on a pair: Abramowitz-Stegun 7.1.26 erfc (|err| <= 1.5e-7), hc = 0.5 erfc(|x| / sqrt2),
// h = x hc + max(x - 2 x hc, 0); ss += h^2
__device__ __forceinline__ void gelu2(float& x0, float& x1, uint64_t& ss2) {
  float t0, t1, e0, e1, q0, q1, d0, d1;
  asm("rcp.approx.ftz.f32 %0, %1;" : "=f"(t0) : "f"(fmaf(fabsf(x0), 0.231641900f, 1.f)));
  asm("rcp.approx.ftz.f32 %0, %1;" : "=f"(t1) : "f"(fmaf(fabsf(x1), 0.231641900f, 1.f)));
  const uint64_t T = f2pack(t0, t1), X = f2pack(x0, x1);
  uint64_t P = fma2(T, f2pack(0.5307027145f, 0.5307027145f), f2pack(-0.7265760135f, -0.7265760135f));
  P = fma2(T, P, f2pack(0.7107068705f, 0.7107068705f));
  P = fma2(T, P, f2pack(-0.142248368f, -0.142248368f));
  P = fma2(T, P, f2pack(0.127414796f, 0.127414796f));
  P = mul2(P, T);
  f2unpack(mul2(X, mul2(X, f2pack(-0.72134752044f, -0.72134752044f))), q0, q1);
  asm("ex2.approx.ftz.f32 %0, %1;" : "=f"(e0) : "f"(q0));
  asm("ex2.approx.ftz.f32 %0, %1;" : "=f"(e1) : "f"(q1));
  const uint64_t XH = mul2(X, mul2(P, f2pack(e0, e1)));
  f2unpack(fma2(XH, f2pack(-2.f, -2.f), X), d0, d1);
  const uint64_t H = add2(XH, f2pack(fmaxf(d0, 0.f), fmaxf(d1, 0.f)));
  ss2 = fma2(H, H, ss2);
  f2unpack(H, x0, x1);
}
// tile t: tokens [t0, t0 + nv); tokens < cb belong to batch b0, the rest to b0 + 1 (HW > 128: at most two batches)
__device__ __forceinline__ void tile_geom(int t, int N, int HW, int& t0, int& nv, int& b0, int& cb) {
  t0 = t * 128;
  nv = min(128, N - t0);
  b0 = t0 / HW;
  cb = min(nv, (b0 + 1) * HW - t0);
}

          if (tok < nvn) asm volatile("prefetch.global.L2 [%0];" ::"l"(p.x + (size_t)(tn0 + tok) * 128 + (tid & 3) * 32));
        }
      }
      if (has2) {
        int t0, nv, b0, cb;
        tile_geom(t2, p.N, p.HW, t0, nv, b0, cb);
        const bool two = nv > cb;
        if (tid == 0) {
          while (ld_acquire(g_ctr + 2 + b0) < (unsigned)p.HW) { }
          if (two) { while (ld_acquire(g_ctr + 3 + b0) < (unsigned)p.HW) { } }
        }
        wbar();
        // ---- GRN scale s = 1 + gw * G / (mean_k G + eps), per batch slot
#pragma unroll 1
        for (int sl = 0; sl < (two ? 2 : 1); sl++) {
          const float g = __fsqrt_rn(__ldcg(p.gsum + (b0 + sl) * 512 + tid));
          const float ws = warp_sum(g);
          if (lane == 0) s_red[warp] = ws;
          wbar();
          float tot = 0.f;
#pragma unroll
          for (int i = 0; i < NW; i++) tot += s_red[i];
          s_sc[sl * 512 + tid] = fmaf(s_gw[tid], __fdiv_rn(g, tot * (1.f / 512.f) + p.eps), 1.f);
          wbar();
        }
        if (!two) s_sc[512 + tid] = s_sc[tid];
        wbar();
        // ---- h' = h * s from the workspace, split hi/lo into two 32 KB MN-major B buffers (64 hidden rows each)
        const float* hprev = p.hws + ((size_t)(((r - 1) & 1) * G + cta) << 16);
#pragma unroll 1
        for (int kc = 0; kc < 8; kc++) {
          const int buf = kc & 1;
          if (kc < 2) {
            if (pend) { mbar_wait(XFREE, (uint32_t)((g1c - 1) & 1)); pend = false; }
          } else {
            mbar_wait(HFREE(buf), (uint32_t)(((kc >> 1) - 1) & 1));
          }
          float4 hv[2][2];
#pragma unroll
          for (int i = 0; i < 2; i++) {
            const int gi = tid + 512 * i, kr = gi >> 4, tg = gi & 15;
            const float4* src = reinterpret_cast<const float4*>(hprev + (size_t)(kc * 64 + kr) * 128 + tg * 8);
            hv[i][0] = __ldcg(src);
            hv[i][1] = __ldcg(src + 1);
          }
#pragma unroll
          for (int i = 0; i < 2; i++) {
            const int gi = tid + 512 * i, kr = gi >> 4, tg = gi & 15, k = kc * 64 + kr;
            const float sc0 = s_sc[k], sc1 = s_sc[512 + k];
            const float e[8] = {hv[i][0].x, hv[i][0].y, hv[i][0].z, hv[i][0].w, hv[i][1].x, hv[i][1].y, hv[i][1].z, hv[i][1].w};
            uint32_t hi[4], lo[4];
#pragma unroll
            for (int m = 0; m < 4; m++) {
              const int ta = tg * 8 + 2 * m;
              split2(e[2 * m] * (ta < cb ? sc0 : sc1), e[2 * m + 1] * (ta + 1 < cb ? sc0 : sc1), hi[m], lo[m]);
            }
            const int nc = tg >> 3, uu = tg & 7;
            unsigned char* d = sm + OFF_X + buf * 32768 + nc * 8192 + kr * 128 + ((uu ^ (kr & 7)) << 4);
            *reinterpret_cast<uint4*>(d) = make_uint4(hi[0], hi[1], hi[2], hi[3]);
            *reinterpret_cast<uint4*>(d + 16384) = make_uint4(lo[0], lo[1], lo[2], lo[3]);
          }
          fence_proxy_async();
          wbar();
          if (tid == 0) mbar_arrive(HFULL(buf));
        }
        // ---- out = (D2 main + D2 cross) + b2', lanes along output channels, one token per store instruction
        mbar_wait(G2DONE, (uint32_t)(g2c & 1));
        g2c++;
        tc_after();
        const int c = 32 * q4 + lane;
        const float bb = s_b2p[c];
#pragma unroll
        for (int hf = 0; hf < 2; hf++) {
          const int c0 = 32 * g4 + 16 * hf;
          float m[16], xc[16];
          tmem_ld16(tmem + ((uint32_t)(32 * q4) << 16) + (uint32_t)(256 + c0), m);
          tmem_ld16(tmem + ((uint32_t)(32 * q4) << 16) + (uint32_t)(384 + c0), xc);
          float* op = p.out + (size_t)(t0 + c0) * 128 + c;
#pragma unroll
          for (int e = 0; e < 16; e++)
            if (c0 + e < nv) op[(size_t)e * 128] = (m[e] + xc[e]) + bb;
        }
        tc_before();
        wbar();
        if (tid == 0) mbar_arrive(TFREE2);
      }
    }
  } else if (warp == NW) {
    // ---- MMA warp (uniform loop, elected issue)
    const uint32_t id = idesc_f16(128);
    int q = 0, g1c = 0, g2c = 0;
#pragma unroll 1
    for (int r = 0; r <= R; r++) {
      const bool has1 = r * G + cta < p.ntiles;
      const bool has2 = (r >= 1) && ((r - 1) * G + cta < p.ntiles);
      if (has1) {
        mbar_wait(XFULL, (uint32_t)(g1c & 1));
        tc_after();
#pragma unroll 1
        for (int hb = 0; hb < 4; hb++) {
          const int buf = hb & 1, use = 2 * g1c + (hb >> 1);
          mbar_wait(TFREE(buf), (uint32_t)((use & 1) ^ 1));
          tc_after();
          const uint32_t d = tmem + (uint32_t)(buf * 128);
#pragma unroll 1
          for (int kc = 0; kc < 2; kc++) {
            const int s = q & 3;
            mbar_wait(FULL(s), (uint32_t)((q >> 2) & 1));
            tc_after();
            const uint32_t wa = sb + OFF_W + s * 32768;
#pragma unroll
            for (int j = 0; j < 4; j++) {
              const uint64_t aH = sdesc(wa + j * 32, 16, 1024), aL = sdesc(wa + 16384 + j * 32, 16, 1024);
              const uint32_t bo = (uint32_t)((kc * 64 + j * 16) * 128);
              const uint64_t bH = sdesc(sb + OFF_X + bo, 16384, 1024), bL = sdesc(sb + OFF_X + 32768 + bo, 16384, 1024);
              mma_issue(d, aH, bH, id, (kc | j) ? 1u : 0u);
              mma_issue(d, aH, bL, id, 1u);
              mma_issue(d, aL, bH, id, 1u);
            }
            mma_commit(EMPTY(s));
            q++;
          }
          mma_commit(G1FULL(buf));
        }
        mma_commit(XFREE);
        g1c++;
      }
      if (has2) {
        mbar_wait(TFREE2, (uint32_t)((g2c & 1) ^ 1));
        tc_after();
        const uint32_t dm = tmem + 256u, dc = tmem + 384u;
#pragma unroll 1
        for (int kc = 0; kc < 8; kc++) {
          const int hbuf = kc & 1;
          mbar_wait(HFULL(hbuf), (uint32_t)((kc >> 1) & 1));
          tc_after();
          const int s = q & 3;
          mbar_wait(FULL(s), (uint32_t)((q >> 2) & 1));
          tc_after();
          const uint32_t wa = sb + OFF_W + s * 32768;
          const uint32_t hbase = sb + OFF_X + hbuf * 32768;
#pragma unroll
          for (int j = 0; j < 4; j++) {
            const uint64_t aH = sdesc(wa + j * 32, 16, 1024), aL = sdesc(wa + 16384 + j * 32, 16, 1024);
            const uint64_t bH = sdesc(hbase + j * 2048, 8192, 1024), bL = sdesc(hbase + 16384 + j * 2048, 8192, 1024);
            const uint32_t first = (kc | j) ? 1u : 0u;
            mma_issue(dm, aH, bH, id, first);
            mma_issue(dc, aH, bL, id, first);
            mma_issue(dc, aL, bH, id, 1u);
          }
          mma_commit(EMPTY(s));
          mma_commit(HFREE(hbuf));
          q++;
        }
        mma_commit(G2DONE);
        g2c++;
      }
    }
  } else {
    // ---- weight producer: 16 stages of 32 KB per tile (8 for GEMM1 of tile r, 8 for GEMM2 of tile r - 1)
    if (lane == 0) {
      while (ld_acquire(g_ctr) < (unsigned)G) { }
      asm volatile("fence.proxy.async;" ::: "memory");
      int q = 0;
#pragma unroll 1
      for (int r = 0; r <= R; r++) {
        const bool has1 = r * G + cta < p.ntiles;
        const bool has2 = (r >= 1) && ((r - 1) * G + cta < p.ntiles);
#pragma unroll 1
        for (int si = 0; si < 16; si++) {
          if ((si < 8 && !has1) || (si >= 8 && !has2)) continue;
          const int s = q & 3;
          mbar_wait(EMPTY(s), (uint32_t)(((q >> 2) & 1) ^ 1));
          mbar_expect(FULL(s), 32768u);
          bulk_g2s(sb + OFF_W + s * 32768, p.wimg + (size_t)si * 32768, 32768u, FULL(s));
          q++;
        }
      }
    }
    __syncwarp();
  }
  tc_before();
  __syncthreads();
  if (tid == 0) {
    const unsigned old = atomicAdd(g_ctr + 1, 1u);
    if (old == (unsigned)G - 1) {
      for (int b = 0; b < p.B; b++) atomicExch(g_ctr + 2 + b, 0u);
      atomicExch(g_ctr, 0u);
      __threadfence();
      atomicExch(g_ctr + 1, 0u);
    }
  }
  if (warp == NW) {
    tc_after();
    asm volatile("tcgen05.dealloc.cta_group::1.sync.aligned.b32 %0, %1;" ::"r"(tmem), "r"(512) : "memory");
  }
}

extern "C" const char* grn_launch(GParams p, void* stream) {
  static bool attr_set = false;
  cudaError_t e;
  if (!attr_set) {  // one maximal value for every shape
    e = cudaFuncSetAttribute(grn_mlp_kernel, cudaFuncAttributeMaxDynamicSharedMemorySize, SMEM_TOTAL);
    if (e != cudaSuccess) return cudaGetErrorString(e);
    attr_set = true;
  }
  if (p.B > 64 || p.HW <= 128) return "grn_launch: unsupported shape";
  void* args[] = {(void*)&p};
  e = cudaLaunchCooperativeKernel((const void*)grn_mlp_kernel, dim3(p.grid), dim3(NTH), args, (size_t)SMEM_TOTAL,
                                  (cudaStream_t)stream);
  if (e != cudaSuccess) return cudaGetErrorString(e);
  return nullptr;
}

#include <torch/extension.h>
#include <ATen/cuda/CUDAContext.h>
#include <c10/cuda/CUDAGuard.h>
#include <algorithm>

struct GParams {
  const float* x; const float* w1; const float* b1; const float* gw; const float* gb; const float* w2; const float* b2;
  float* out; unsigned char* wimg; float* b2p; float* gsum; float* hws;
  int N, HW, B, ntiles, grid; float eps;
};
extern "C" const char* grn_launch(GParams p, void* stream);

void run(torch::Tensor x, torch::Tensor w1, torch::Tensor b1, torch::Tensor gw, torch::Tensor gb, torch::Tensor w2,
         torch::Tensor b2, double eps, torch::Tensor out) {
  for (auto* t : {&x, &w1, &b1, &gw, &gb, &w2, &b2, &out}) {
    TORCH_CHECK(t->is_cuda() && t->is_contiguous() && t->scalar_type() == torch::kFloat32, "fp32 contiguous CUDA tensors");
  }
  TORCH_CHECK(x.dim() == 4 && x.size(3) == 128 && w1.size(0) == 512 && w2.size(0) == 128, "shape");
  const c10::cuda::CUDAGuard guard(x.device());
  const int B = (int)x.size(0), HW = (int)(x.size(1) * x.size(2)), N = B * HW;
  const int ntiles = (N + 127) / 128;
  const int sms = at::cuda::getCurrentDeviceProperties()->multiProcessorCount;
  const int grid = std::min(sms, ntiles);
  // workspace (uninitialised; every byte read in a call is written earlier in that call):
  // weight images 512 KB | b2' 512 B | gsum B x 512 | h ring 2 x grid x 256 KB
  const size_t off_b2p = 524288, off_gsum = off_b2p + 1024;
  const size_t off_h = (off_gsum + (size_t)B * 2048 + 1023) & ~(size_t)1023;
  const size_t total = off_h + (size_t)2 * grid * 65536 * 4;
  auto ws = torch::empty({(int64_t)total}, x.options().dtype(torch::kUInt8));
  unsigned char* base = ws.data_ptr<unsigned char>();
  GParams p;
  p.x = x.data_ptr<float>(); p.w1 = w1.data_ptr<float>(); p.b1 = b1.data_ptr<float>(); p.gw = gw.data_ptr<float>();
  p.gb = gb.data_ptr<float>(); p.w2 = w2.data_ptr<float>(); p.b2 = b2.data_ptr<float>(); p.out = out.data_ptr<float>();
  p.wimg = base; p.b2p = (float*)(base + off_b2p); p.gsum = (float*)(base + off_gsum); p.hws = (float*)(base + off_h);
  p.N = N; p.HW = HW; p.B = B; p.ntiles = ntiles; p.grid = grid; p.eps = (float)eps;
  const char* err = grn_launch(p, (void*)at::cuda::getCurrentCUDAStream().stream());
  TORCH_CHECK(err == nullptr, err ? err : "");
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) { m.def("run", &run); }
